  1230→export function useFilteredChats() {
  1231→  return useApp(
  1232→    useShallow((s) => {
  1233→      const q = s.searchQuery.trim().toLowerCase()
  1234→      let list = s.chats
  1235→      if (q) {
  1236→        list = list.filter(
  1237→          (c) =>
  1238→            c.name.toLowerCase().includes(q) ||
  1239→            (c.lastMessage?.text ?? '').toLowerCase().includes(q),
  1240→        )
  1241→      }
  1242→      // Saved first, then by last message desc
  1243→      return [...list].sort((a, b) => {
  1244→        if (a.kind === 'saved' && b.kind !== 'saved') return -1
  1245→        if (b.kind === 'saved' && a.kind !== 'saved') return 1
  1246→        const at = (c: typeof a) => c.lastMessage ? Date.parse(c.lastMessage.createdAt) : Date.parse(c.createdAt)
  1247→        return at(b) - at(a)
  1248→      })
  1249→    }),
  1250→  )
  1251→}
